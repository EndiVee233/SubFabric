cmake_minimum_required(VERSION 3.22)
project(subfabric_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${CRISP_SOURCE}/ggml ggml)
find_package(Threads REQUIRED)
add_library(crispasr-core STATIC ${CRISP_SOURCE}/src/core/mel.cpp ${CRISP_SOURCE}/src/core/gguf_loader.cpp ${CRISP_SOURCE}/src/core/audio_resample.cpp ${CRISP_SOURCE}/src/crispasr_imatrix.cpp)
target_include_directories(crispasr-core PUBLIC ${CRISP_SOURCE}/src ${CRISP_SOURCE}/include ${CRISP_SOURCE}/ggml/src)
target_link_libraries(crispasr-core PUBLIC ggml ggml-base Threads::Threads)
add_subdirectory(${CRISP_SOURCE}/crisp_audio crisp_audio)
add_library(subfabric_asr SHARED bridge.cpp ${CRISP_SOURCE}/src/parakeet.cpp ${CRISP_SOURCE}/src/parakeet_orchestrate.cpp ${CRISP_SOURCE}/src/qwen3_asr.cpp ${CRISP_SOURCE}/src/core/qwen3_ultra_sidecar.cpp)
target_include_directories(subfabric_asr PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${Vulkan_INCLUDE_DIR} ${CRISP_SOURCE}/src ${CRISP_SOURCE}/include ${CRISP_SOURCE}/ggml/src)
target_compile_definitions(subfabric_asr PRIVATE GGML_USE_VULKAN)
target_link_libraries(subfabric_asr PRIVATE crispasr-core crisp_audio ggml ggml-vulkan vulkan log)
target_link_options(subfabric_asr PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
target_include_directories(ggml-vulkan PRIVATE ${CRISP_SOURCE}/../spirv-headers/include)
